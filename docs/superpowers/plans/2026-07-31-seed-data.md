# Seed Data Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Seed 27 tình huống pháp luật thật (9 mỗi chủ đề Giao thông / Mạng xã hội / Sở hữu trí tuệ) vào D1 dưới dạng đồ thị 4 bảng đã duyệt bốn mắt, và cho chat hiện mức phạt + căn cứ khi entry có citation bốn mắt.

**Architecture:** File dữ liệu thuần (`db/seeds/seed-content.v1.mjs`) → script (`scripts/seed-d1.mjs`) validate + tính checksum bằng chính `computeProvisionChecksum` của app → sinh SQL idempotent trong một transaction → áp vào D1 local qua `node:sqlite` hoặc xuất file cho production. Một thay đổi code duy nhất trong `lib/legal-chat.ts`.

**Tech Stack:** Node ≥22.13 (node:sqlite, --experimental-strip-types), Drizzle ORM (đã có), node --test, không thêm dependency mới.

## Global Constraints

- Topic chỉ nhận đúng 3 giá trị: `"Giao thông"`, `"Mạng xã hội"`, `"Sở hữu trí tuệ"` (TOPICS trong `lib/catalog-resolver.ts`).
- `legal_basis` của entries KHÔNG được chứa Nghị định 131/2013 dưới bất kỳ dạng viết nào (`hasBlockedLegalBasis` trong `lib/legal-content.ts:125` chặn regex `(?:nghidinh|nd)(?:so)?1312013`).
- URL nguồn: https, host thuộc {vbpl.vn, vbpl.moj.gov.vn, chinhphu.vn, *.chinhphu.vn} (CHECK của `legal_sources`).
- Principal seed: `created_by='seed-editor'`, `reviewed_by/verified_by='seed-reviewer'` (bốn mắt yêu cầu hai giá trị khác nhau).
- Checksum: `checksum_version='provision-sha256-v1'`; thuật toán phải là `computeProvisionChecksum` import từ `lib/legal-evidence-retriever.ts` — không tự cài lại.
- Không sửa file fixture (`fixtures/`), không sửa migration đã có, không tắt trigger/PRAGMA khi áp seed.
- Commit chỉ gồm file của task đang làm (repo đang có thay đổi chưa commit ở `lib/openai-*.ts` — không gom vào).
- Mỗi commit theo format `<type>: <description>`, không trailer.

function d1Adapter(sqlite) {
  return {
    prepare(query) {
      const bound = [];
      const statement = {
        bind(...values) { bound.push(...values); return statement; },
        run() {
          const rows = /^\s*(select|insert.*returning)/is.test(query)
            ? sqlite.prepare(query).all(...bound)
            : (sqlite.prepare(query).run(...bound), []);
          return { success: true, results: rows };
        },
        _exec() { return statement.run(); },
      };
      return statement;
    },
    async batch(statements) { return statements.map((s) => s._exec()); },
  };
}

export function validateSeedContent(content) {
  const errors = [];
  const slugs = new Set();
  if (content?.seedContentVersion !== "seed-content-v1") {
    errors.push("seedContentVersion khong dung");
  }
  for (const s of content?.situations ?? []) {
    const where = s.slug ?? "<khong slug>";
    if (!s.slug || slugs.has(s.slug)) errors.push(`${where}: slug trung/thieu`);
    slugs.add(s.slug);
    if (!TOPICS.has(s.topic)) errors.push(`${where}: topic khong hop le`);
    if (hasBlockedLegalBasis(s.legalBasis ?? "")) {
      errors.push(`${where}: legalBasis dinh blocklist ND 131/2013`);
    }
    for (const field of ["title", "legalBasis", "penalty", "remedy", "caseStudy", "icon"]) {
      if (typeof s[field] !== "string" || !s[field].trim()) {
        errors.push(`${where}: thieu ${field}`);
      }
    }
    if (!Array.isArray(s.tags) || s.tags.length === 0) {
      errors.push(`${where}: thieu tags`);
    }
    let url;
    try {
      url = new URL(s.source?.officialUrl ?? "");
    } catch {
      errors.push(`${where}: officialUrl khong phai URL`);
    }
    if (url && (url.protocol !== "https:" || !isAllowedHost(url.hostname))) {
      errors.push(`${where}: officialUrl ngoai allowlist`);
    }
    for (const field of ["documentNumber", "title", "issuedAt", "effectiveFrom"]) {
      if (typeof s.source?.[field] !== "string" || !s.source[field].trim()) {
        errors.push(`${where}: source thieu ${field}`);
      }
    }
    for (const field of ["originalText", "simplifiedText"]) {
      if (typeof s.provision?.[field] !== "string" || !s.provision[field].trim()) {
        errors.push(`${where}: provision thieu ${field}`);
      }
    }
  }
  return errors;
}

const q = (value) =>
  value === null || value === undefined
    ? "NULL"
    : `'${String(value).replace(/'/g, "''")}'`;

export async function buildSeedSql(content, { now = () => new Date() } = {}) {
  const errors = validateSeedContent(content);
  if (errors.length) {
    throw new Error(`Noi dung seed khong hop le:\n${errors.join("\n")}`);
  }
  const stamp = now().toISOString();
  const lines = ["BEGIN;"];

  const sources = new Map();
  for (const s of content.situations) {
    if (!sources.has(s.source.documentNumber)) {
      sources.set(s.source.documentNumber, s.source);
    }
  }
  for (const src of sources.values()) {
    const host = new URL(src.officialUrl).hostname.toLowerCase();
    lines.push(`
INSERT INTO legal_sources (
  document_number, title, official_url, official_host, issued_at,
  effective_from, status, created_by, last_verified_at, verified_by
)
SELECT ${q(src.documentNumber)}, ${q(src.title)}, ${q(src.officialUrl)},
  ${q(host)}, ${q(src.issuedAt)}, ${q(src.effectiveFrom)}, 'in_force',
  ${q(SEED_EDITOR)}, ${q(stamp)}, ${q(SEED_REVIEWER)}
WHERE NOT EXISTS (
  SELECT 1 FROM legal_sources WHERE document_number = ${q(src.documentNumber)}
);`);
  }

  for (const s of content.situations) {
    const revisionId = `seed-${s.slug}-v1`;
    const checksum = await computeProvisionChecksum({
      source: {
        documentNumber: s.source.documentNumber,
        officialUrl: s.source.officialUrl,
      },
      provision: {
        revisionId,
        checksumVersion: PROVISION_CHECKSUM_VERSION,
        article: s.provision.article ?? null,
        clause: s.provision.clause ?? null,
        point: s.provision.point ?? null,
        originalText: s.provision.originalText,
        simplifiedText: s.provision.simplifiedText,
        effectivityStatus: "in_force",
        effectiveFrom: s.source.effectiveFrom,
        effectiveTo: null,
      },
    });
    lines.push(`
INSERT INTO legal_provisions (
  source_id, article, clause, point, original_text, simplified_text,
  status, created_by, reviewed_by, reviewed_at, revision_id,
  checksum_version, checksum_sha256, effectivity_status, effective_from
)
SELECT s.id, ${q(s.provision.article ?? null)}, ${q(s.provision.clause ?? null)},
  ${q(s.provision.point ?? null)}, ${q(s.provision.originalText)},
  ${q(s.provision.simplifiedText)}, 'published', ${q(SEED_EDITOR)},
  ${q(SEED_REVIEWER)}, ${q(stamp)}, ${q(revisionId)},
  ${q(PROVISION_CHECKSUM_VERSION)}, ${q(checksum)}, 'in_force',
  ${q(s.source.effectiveFrom)}
FROM legal_sources s
WHERE s.document_number = ${q(s.source.documentNumber)}
  AND NOT EXISTS (
    SELECT 1 FROM legal_provisions WHERE revision_id = ${q(revisionId)}
  );`);
    lines.push(`
INSERT INTO legal_entries (
  topic, icon, title, legal_basis, penalty, remedy, case_study, tags,
  status, review_status, created_by, reviewed_by, reviewed_at
)
SELECT ${q(s.topic)}, ${q(s.icon)}, ${q(s.title)}, ${q(s.legalBasis)},
  ${q(s.penalty)}, ${q(s.remedy)}, ${q(s.caseStudy)},
  ${q(JSON.stringify(s.tags))}, 'published', 'four_eyes_verified',
  ${q(SEED_EDITOR)}, ${q(SEED_REVIEWER)}, ${q(stamp)}
WHERE NOT EXISTS (
  SELECT 1 FROM legal_entries
  WHERE topic = ${q(s.topic)} AND title = ${q(s.title)}
);`);
    lines.push(`
INSERT INTO legal_entry_citations (
  legal_entry_id, provision_id, display_order, review_status,
  created_by, reviewed_by, reviewed_at, cited_revision_id,
  cited_checksum_version, cited_checksum_sha256
)
SELECT e.id, p.id, 0, 'four_eyes_verified', ${q(SEED_EDITOR)},
  ${q(SEED_REVIEWER)}, ${q(stamp)}, ${q(revisionId)},
  ${q(PROVISION_CHECKSUM_VERSION)}, ${q(checksum)}
FROM legal_entries e, legal_provisions p
WHERE e.topic = ${q(s.topic)} AND e.title = ${q(s.title)}
  AND p.revision_id = ${q(revisionId)}
  AND NOT EXISTS (
    SELECT 1 FROM legal_entry_citations c
    WHERE c.legal_entry_id = e.id AND c.provision_id = p.id
  );`);
  }

  lines.push("COMMIT;");
  return lines.join("\n");
}

async function fetchVerifiedEntryCitations(
  db: Awaited<ReturnType<typeof getInitializedDb>>,
  entryId: number,
): Promise<ReviewedCitationPresentationInput[]> {
  const rows = await db
    .select({
      title: legalSources.title,
      documentNumber: legalSources.documentNumber,
      issuedAt: legalSources.issuedAt,
      article: legalProvisions.article,
      clause: legalProvisions.clause,
      point: legalProvisions.point,
      effectiveFrom: legalProvisions.effectiveFrom,
      effectiveTo: legalProvisions.effectiveTo,
      lastVerifiedAt: legalSources.lastVerifiedAt,
    })
    .from(legalEntryCitations)
    .innerJoin(
      legalProvisions,
      eq(legalEntryCitations.provisionId, legalProvisions.id),
    )
    .innerJoin(legalSources, eq(legalProvisions.sourceId, legalSources.id))
    .where(
      and(
        eq(legalEntryCitations.legalEntryId, entryId),
        eq(legalEntryCitations.reviewStatus, "four_eyes_verified"),
        eq(legalProvisions.status, "published"),
        eq(legalSources.status, "in_force"),
        isNotNull(legalSources.lastVerifiedAt),
        isNotNull(legalSources.verifiedBy),
        eq(
          legalEntryCitations.citedChecksumSha256,
          legalProvisions.checksumSha256,
        ),
      ),
    )
    .orderBy(legalEntryCitations.displayOrder)
    .limit(8);
  return rows.flatMap((row) =>
    row.issuedAt && row.effectiveFrom && row.lastVerifiedAt
      ? [
          {
            title: row.title,
            documentNumber: row.documentNumber,
            issuedAt: row.issuedAt,
            article: row.article ?? undefined,
            clause: row.clause ?? undefined,
            point: row.point ?? undefined,
            effectiveFrom: row.effectiveFrom,
            effectiveTo: row.effectiveTo ?? undefined,
            lastVerifiedAt: row.lastVerifiedAt,
          },
        ]
      : [],
  );
}
